# U-Boot boot script for SO3-on-AVZ on the Toradex Verdin iMX8MP.
#
# Loads the single AVZ+SO3 FIT image and boots it. bootm parses the FIT and
# copies each sub-image to its load address (AVZ -> 0x40080000, SO3 guest ->
# 0x48000000, dtbs, rootfs.fat), then enters AVZ which in turn starts the SO3
# guest. The ITB is loaded high (0x90000000) so it never overlaps the
# 0x40080000-0x50c00000 sub-image range during extraction.

echo "=== SO3 on AVZ (Verdin iMX8MP) ==="

env set itb_addr 0x90000000

ext4load mmc 2:1 ${itb_addr} /verdin_imx8mp_avz_so3.itb
bootm ${itb_addr}
