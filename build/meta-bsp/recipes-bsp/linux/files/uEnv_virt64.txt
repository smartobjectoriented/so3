fdt_high=ffffffffffffffff
initrd_high=ffffffffffffffff

# Bare bsp-linux on virt64: U-Boot bootm's a single ITB carrying kernel
# + DTB + buildroot initrd. No AVZ. QEMU loads U-Boot directly via
# -bios filesystem/u-boot.bin (no flash0.img in this mode).

LOAD_ADDR=0x65000000

start=load virtio 0:1 ${LOAD_ADDR} virt64.itb; bootm ${LOAD_ADDR}

bootargs=earlycon=pl011,mmio32,0x09000000 console=ttyAMA0,115200n8 root=/dev/ram rw
